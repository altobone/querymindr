 *
 *
 * CHECKING KEY STOCK:
 *    In wp-cli: wp eval "echo count(get_option('querymindr_license_keys', [])) . ' keys remaining';"
 *    Or check WooCommerce → Orders → any Querymindr order to see assigned key.
 */

define( 'QUERYMINDR_PRODUCTS', [ 'querymindr-mac', 'querymindr-windows', 'querymindr-linux' ] );

/**
 * Assign a license key when an order is completed.
 */
add_action( 'woocommerce_order_status_completed', 'querymindr_assign_license_key' );
function querymindr_assign_license_key( $order_id ) {
	$order = wc_get_order( $order_id );
	if ( ! $order ) return;

	if ( $order->get_meta( '_querymindr_license_key' ) ) return;

	$has_querymindr = false;
	foreach ( $order->get_items() as $item ) {
		$product = $item->get_product();
		if ( $product && in_array( $product->get_slug(), QUERYMINDR_PRODUCTS ) ) {
			$has_querymindr = true;
			break;
		}
	}
	if ( ! $has_querymindr ) return;

	$keys = get_option( 'querymindr_license_keys', [] );
	if ( empty( $keys ) ) {
		error_log( 'Querymindr WARNING: License key pool is empty! Order #' . $order_id . ' has no key.' );
		$order->add_order_note( '⚠️ Querymindr key pool empty — customer needs a key manually.' );
		$order->save();
		return;
	}

	$key = array_shift( $keys );
	update_option( 'querymindr_license_keys', $keys );

	$order->update_meta_data( '_querymindr_license_key', $key );
	$order->add_order_note( 'Querymindr license key assigned: ' . $key );
	$order->save();
}

/**
 * Inject license key into the customer order confirmation email.
 */
add_action( 'woocommerce_email_order_details', 'querymindr_add_key_to_email', 5, 4 );
function querymindr_add_key_to_email( $order, $sent_to_admin, $plain_text, $email ) {
	if ( $sent_to_admin ) return;
	if ( ! in_array( $email->id, [ 'customer_completed_order', 'customer_invoice' ] ) ) return;

	$key = $order->get_meta( '_querymindr_license_key' );
	if ( ! $key ) return;

	if ( $plain_text ) {
		echo "\n\nYOUR QUERYMINDR LICENSE KEY\n";
		echo "----------------------------\n";
		echo $key . "\n";
		echo "\nTo activate: open Querymindr → Settings → paste your key → click Activate.\n\n";
	} else {
		echo '<div style="margin:24px 0;padding:24px;background:#f5f3ff;border-left:4px solid #4f46e5;border-radius:4px;">';
		echo '<p style="margin:0 0 8px;font-size:13px;font-weight:600;text-transform:uppercase;letter-spacing:1px;color:#4f46e5;">Your Querymindr License Key</p>';
		echo '<p style="margin:0 0 12px;font-size:24px;font-weight:700;letter-spacing:3px;color:#1e1b4b;font-family:monospace;">' . esc_html( $key ) . '</p>';
		echo '<p style="margin:0;font-size:13px;color:#555;">To activate: open Querymindr → click <strong>Settings</strong> → paste your key → click <strong>Activate</strong>.</p>';
		echo '</div>';
	}
}

/**
 * Show the assigned key inside the WooCommerce admin order screen.
 */
add_action( 'woocommerce_admin_order_data_after_billing_address', 'querymindr_show_key_in_admin' );
function querymindr_show_key_in_admin( $order ) {
	$key = $order->get_meta( '_querymindr_license_key' );
	if ( ! $key ) return;
	echo '<p style="margin-top:12px;"><strong>Querymindr Key:</strong> <code>' . esc_html( $key ) . '</code></p>';

	$remaining = count( get_option( 'querymindr_license_keys', [] ) );
	echo '<p style="font-size:12px;color:#888;">' . $remaining . ' keys remaining in pool</p>';
}
